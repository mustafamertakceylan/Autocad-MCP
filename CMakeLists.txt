cmake_minimum_required(VERSION 3.25)
project(AutocadMcp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing pinned nlohmann json.hpp")
if(NOT EXISTS "${JSON_INCLUDE_DIR}/json.hpp")
  set(JSON_INCLUDE_DIR "${CMAKE_BINARY_DIR}/dependencies")
  file(MAKE_DIRECTORY "${JSON_INCLUDE_DIR}")
  file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.12.0/single_include/nlohmann/json.hpp"
    "${JSON_INCLUDE_DIR}/json.hpp"
    EXPECTED_HASH SHA256=aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63 TLS_VERIFY ON)
endif()
add_library(bridge_core STATIC native/bridge.cpp)
set_target_properties(bridge_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(bridge_core PUBLIC native "${JSON_INCLUDE_DIR}")
if(WIN32)
  target_compile_options(bridge_core PUBLIC /utf-8 /W4 /permissive-)
  target_compile_definitions(bridge_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(bridge_core PUBLIC bcrypt advapi32 crypt32)
elseif(APPLE)
  find_package(Threads REQUIRED)
  target_link_libraries(bridge_core PUBLIC Threads::Threads "-framework Security")
else()
  message(FATAL_ERROR "This prototype targets Windows and macOS only")
endif()
add_executable(bridge_core_test native/core_test.cpp)
target_link_libraries(bridge_core_test PRIVATE bridge_core)
enable_testing()
add_test(NAME bridge_core COMMAND bridge_core_test)
set(OBJECTARX_ROOT "" CACHE PATH "Licensed ObjectARX 2027 SDK root")
option(AUTOCAD_MCP_TESTING "Compile fixed synthetic desktop acceptance controls (never ship)" OFF)
if(OBJECTARX_ROOT)
  if(NOT EXISTS "${OBJECTARX_ROOT}/inc/aced.h")
    message(FATAL_ERROR "OBJECTARX_ROOT must contain inc/aced.h")
  endif()
  add_library(AutocadMcp MODULE native/plugin.cpp native/plugin.def)
  file(GLOB MCP_NATIVE_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/native/*.cpp" "${CMAKE_SOURCE_DIR}/native/*.hpp" "${CMAKE_SOURCE_DIR}/native/*.def")
  set(MCP_NATIVE_HASHES "")
  foreach(MCP_SOURCE IN LISTS MCP_NATIVE_SOURCES)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MCP_SOURCE}")
    file(SHA256 "${MCP_SOURCE}" MCP_SOURCE_HASH)
    string(APPEND MCP_NATIVE_HASHES "${MCP_SOURCE_HASH}")
  endforeach()
  string(SHA256 MCP_NATIVE_BUILD_ID "${PROJECT_VERSION};${MCP_NATIVE_HASHES};${AUTOCAD_MCP_TESTING}")
  target_compile_definitions(AutocadMcp PRIVATE "AUTOCAD_MCP_BUILD_ID=\"${MCP_NATIVE_BUILD_ID}\"")
  target_include_directories(AutocadMcp PRIVATE "${OBJECTARX_ROOT}/inc" "${OBJECTARX_ROOT}/inc-x64")
  target_compile_definitions(AutocadMcp PRIVATE _ACRXAPP ACRXAPP)
  if(AUTOCAD_MCP_TESTING)
    target_compile_definitions(AutocadMcp PRIVATE AUTOCAD_MCP_TESTING)
  endif()
  target_link_libraries(AutocadMcp PRIVATE bridge_core)
  if(WIN32)
    target_link_directories(AutocadMcp PRIVATE "${OBJECTARX_ROOT}/lib-x64")
    target_link_libraries(AutocadMcp PRIVATE rxapi acdb26 acge26 acgeoment ac1st26 AcPal acad accore AcPublish_crx)
    target_compile_options(AutocadMcp PRIVATE /W4 /utf-8 /permissive-)
    set_target_properties(AutocadMcp PROPERTIES PREFIX "" SUFFIX ".arx")
    add_custom_command(TARGET AutocadMcp POST_BUILD
      COMMAND "${CMAKE_COMMAND}" "-DBINARY=$<TARGET_FILE:AutocadMcp>"
        "-DTESTING=${AUTOCAD_MCP_TESTING}" "-DVERSION=${PROJECT_VERSION}" "-DBUILD_ID=${MCP_NATIVE_BUILD_ID}"
        -P "${CMAKE_SOURCE_DIR}/cmake/native-manifest.cmake"
      VERBATIM)
  elseif(APPLE)
    message(FATAL_ERROR "Mac SDK link layout must be verified on the authorized Mac; no untested native binary is produced")
  endif()
endif()
